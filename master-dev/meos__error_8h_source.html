86extern void meos_error(int errlevel, int errcode, const char *format, ...)
87/*
88 * Give meos_error printf-style format checking. In the PostgreSQL-backed
89 * build pg_attribute_printf is available and selects the gnu_printf
90 * archetype (correct on MinGW, where the default printf archetype is MS
91 * semantics). Clang is excluded from that path because it rejects the
92 * gnu_printf archetype with -Wignored-attributes; it takes the plain
93 * printf archetype below instead. This public header is also consumed
94 * standalone, where the macro may not be in scope, so fall back to the
95 * gnu_printf attribute directly under GCC, plain printf under Clang, and
96 * to nothing on compilers that lack it.
97 */
98#if defined(pg_attribute_printf) && !defined(__clang__)
99 pg_attribute_printf(3, 4);
100#elif defined(__GNUC__) && !defined(__clang__)
101 __attribute__((format(gnu_printf, 3, 4)));
102#elif defined(__GNUC__)
103 __attribute__((format(printf, 3, 4)));
104#else
105 ;
106#endif
