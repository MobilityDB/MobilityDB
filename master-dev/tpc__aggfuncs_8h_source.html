57static inline TInstant **
58tpcpatch_transform_tnpoints(const Temporal *temp, int *count)
59{
60 int n = temporal_num_instants(temp);
61 TInstant **result = palloc(sizeof(TInstant *) * n);
62 for (int i = 0; i < n; i++)
63 {
64 const TInstant *inst = temporal_instant_n(temp, i + 1);
65 const Pcpatch *pa = (const Pcpatch *) DatumGetPointer(
66 tinstant_value_p(inst));
67 int32 npts = pcpatch_npoints(pa);
68 result[i] = tinstant_make(Int32GetDatum(npts), T_TINT, inst->t);
69 }
70 *count = n;
71 return result;
72}

81static inline TInstant **
82tpcpatch_transform_tdensity(const Temporal *temp, int *count)
83{
84 int n = temporal_num_instants(temp);
85 TInstant **result = palloc(sizeof(TInstant *) * n);
86 for (int i = 0; i < n; i++)
87 {
88 const TInstant *inst = temporal_instant_n(temp, i + 1);
89 const Pcpatch *pa = (const Pcpatch *) DatumGetPointer(
90 tinstant_value_p(inst));
91 double xrange = pa->bounds[1] - pa->bounds[0];
92 double yrange = pa->bounds[3] - pa->bounds[2];
93 double area = xrange * yrange;
94 double density = (area > 0.0) ? pa->npoints / area : pa->npoints / 0.0;
95 result[i] = tinstant_make(Float8GetDatum(density), T_TFLOAT, inst->t);
96 }
97 *count = n;
98 return result;
99}
