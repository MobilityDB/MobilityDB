74/* The durations, 2^52 microseconds, below which the offsets of a crossing
75 * and their halves are doubles exactly */
76#define FLOATSEGM_FILTER_MAXD (INT64CONST(1) << 52)
77
98static inline bool
99floatsegm_offset_filter(double x, int64 d, int64 *m)
100{
101 /* Not a number fails both comparisons */
102 double y = x + 0.5;
103 if (! (y >= 1.0 && y < (double) d))
104 return false;
105 TimestampTz est = (TimestampTz) y;
106 if (fabs(x - (double) est) + x * 0x1p-48 > 0.5)
107 return false;
108 *m = est;
109 return true;
110}
111
134static inline bool
135floatsegm_crossing_filter(double start1, double end1, double start2,
136 double end2, int64 d, int64 *m)
137{
138 if (d >= FLOATSEGM_FILTER_MAXD)
139 return false;
140 double a = start1 - start2, b = end1 - end2;
141 return floatsegm_offset_filter((a / (a - b)) * (double) d, d, m);
142}
