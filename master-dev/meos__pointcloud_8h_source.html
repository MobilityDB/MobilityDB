81/* A TPCBox begins with a whole STBox: every STBox field sits at the same
82 * offset, so the shared spatiotemporal code reads a TPCBox through an
83 * @p STBox pointer, and a TPCBox is projected to an STBox by dropping the
84 * trailing pcid. The layout-compatibility cast (CREATE CAST tpcbox AS
85 * stbox) relies on this, as does the SRID accessor.
86 *
87 * ⛔ The flags offset is the one that matters most: nearly every box
88 * operation reads the X / Z / T / GEODETIC bits, so a divergence there
89 * silently misreads every box rather than failing loudly. */
90static_assert(offsetof(TPCBox, period) == offsetof(STBox, period),
91 "TPCBox and STBox must share the period offset");
92static_assert(offsetof(TPCBox, xmin) == offsetof(STBox, xmin),
93 "TPCBox and STBox must share the xmin offset");
94static_assert(offsetof(TPCBox, ymin) == offsetof(STBox, ymin),
95 "TPCBox and STBox must share the ymin offset");
96static_assert(offsetof(TPCBox, zmin) == offsetof(STBox, zmin),
97 "TPCBox and STBox must share the zmin offset");
98static_assert(offsetof(TPCBox, xmax) == offsetof(STBox, xmax),
99 "TPCBox and STBox must share the xmax offset");
100static_assert(offsetof(TPCBox, ymax) == offsetof(STBox, ymax),
101 "TPCBox and STBox must share the ymax offset");
102static_assert(offsetof(TPCBox, zmax) == offsetof(STBox, zmax),
103 "TPCBox and STBox must share the zmax offset");
104static_assert(offsetof(TPCBox, srid) == offsetof(STBox, srid),
105 "TPCBox and STBox must share the srid offset");
106static_assert(offsetof(TPCBox, flags) == offsetof(STBox, flags),
107 "TPCBox and STBox must share the flags offset");
108static_assert(offsetof(TPCBox, pcid) >= sizeof(STBox),
109 "the pcid of a TPCBox must follow the whole STBox prefix");
