86static inline bool
87ensure_same_span_type(const Span *s1, const Span *s2)
88{
89 if (s1->spantype == s2->spantype)
90 return true;
91 meos_error(ERROR, MEOS_ERR_INVALID_ARG_TYPE,
92 "Operation on mixed span types: %s and %s",
93 meostype_name(s1->spantype), meostype_name(s2->spantype));
94 return false;
95}
