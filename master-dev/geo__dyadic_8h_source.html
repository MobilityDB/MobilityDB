145static inline int
146cross_product_sign(double ax, double ay, double bx, double by, double cx,
147 double cy, double dx, double dy)
148{
149 double l1 = bx - ax, l2 = dy - cy, r1 = by - ay, r2 = dx - cx;
150 double left = l1 * l2, right = r1 * r2;
151 double mag = fabs(left) + fabs(right);
152 double bound = (3.0 + 16.0 * DBL_EPSILON) * DBL_EPSILON * mag;
153 if (mag >= 0x1p-969)
154 {
155 double det = left - right;
156 if (det > bound)
157 return 1;
158 if (det < - bound)
159 return -1;
160 }
161 if ((l1 == 0.0 || l2 == 0.0) && (r1 == 0.0 || r2 == 0.0))
162 return 0;
163 return cross_product_sign_exact(ax, ay, bx, by, cx, cy, dx, dy);
164}

679static inline void
680two_square(double a, double *x, double *y)
681{
682 double t = 134217729.0 * a;
683 double ah = t - (t - a);
684 double al = a - ah;
685 *x = a * a;
686 *y = ((ah * ah - *x) + 2.0 * ah * al) + al * al;

695static inline int
696grow_expansion(int elen, const double *e, double b, double *h)
697{
698 double q = b;
699 int hlen = 0;
700 for (int i = 0; i < elen; i++)
701 {
702 double sum = q + e[i];
703 double bv = sum - q;
704 double av = sum - bv;
705 double err = (q - av) + (e[i] - bv);
706 q = sum;
707 if (err != 0.0)
708 h[hlen++] = err;
709 }
710 if (q != 0.0 || hlen == 0)
711 h[hlen++] = q;
712 return hlen;
713}

766static pg_attribute_always_inline DDBound
767ddb_mul(DDBound a, DDBound b)
768{
769 DDBound r;
770 if (ddb_is_zero(a) || ddb_is_zero(b))
771 {
772 r.hi = r.lo = r.e = 0.0;
773 return r;
774 }
775 if (ddb_is_double(a) && ddb_is_double(b))
776 {
777 two_product(a.hi, b.hi, &r.hi, &r.lo);
778 /* A product that underflows is not split exactly */
779 r.e = (fabs(r.hi) < DDB_EXACT_PRODUCT) ? DDB_TINY : 0.0;
780 return r;
781 }
782 double p, pe;
783 two_product(a.hi, b.hi, &p, &pe);
784 pe += a.hi * b.lo + a.lo * b.hi;
785 two_sum(p, pe, &r.hi, &r.lo);
786 double ma = fabs(a.hi) + fabs(a.lo), mb = fabs(b.hi) + fabs(b.lo);
787 r.e = ma * b.e + mb * a.e + a.e * b.e + fabs(a.lo * b.lo) +
788 DDB_EPS * fabs(p) + DDB_TINY;
789 return r;
790}

798 /* A sum with an exact zero is the other term, and the sum of two doubles

863static inline int
864polynomial_sign_filter(const ExactSum *sums, const PolyTerm *terms,
865 int nterms)
866{
867 /* The value in doubles and a bound on its error */
868 double val = 0.0, err = 0.0;
869 for (int i = 0; i < nterms; i++)
870 {
871 double v = terms[i].coef, e = 0.0;
872 for (int k = 0; k < terms[i].deg; k++)
873 {
874 const ExactSum *s = &sums[terms[i].sum[k]];
875 double nv = v * s->approx;
876 e = fabs(v) * s->err + fabs(s->approx) * e + e * s->err +
877 DBL_EPSILON * fabs(nv);
878 v = nv;
879 }
880 double nval = val + v;
881 err += e + DBL_EPSILON * fabs(nval);
882 val = nval;
883 }
884 /* The bound itself is computed in doubles, hence the margin; the absolute
885 * term covers a product that underflows */
886 double bound = err * (1.0 + 1e-6) + DBL_MIN;
887 if (isfinite(val) && isfinite(bound) && fabs(val) > bound)
888 return (val > 0.0) ? 1 : -1;
889 return polynomial_sign_pairs(sums, terms, nterms);
890}

981static inline bool
982nearest_double_search(uint64 bits, NearestBitsSide side,
983 void *arg, int budget, uint64 *result)
984{
985 int asked = 0;
986#define NEAREST_ASK(b, res) \
987 do { \
988 if (budget > 0 && asked++ >= budget) \
989 return false; \
990 int s_ = side((b), arg); \
991 if (s_ == 2) \
992 (res) = 2; \
993 else \
994 (res) = nearest_bits_beyond_sign(s_, (b)) ? 1 : 0; \
995 } while (0)
996 const uint64 max_bits = NEAREST_INF_BITS;
997 uint64 lo, hi, step = 1;
998 int beyond, moves = 0;
999 if (bits > max_bits)
1000 bits = max_bits;
1001 if (bits == max_bits)
1002 beyond = 0;
1003 else
1004 NEAREST_ASK(bits, beyond);
1005 if (beyond == 2)
1006 {
1007 /* The candidate cannot tell: below its lower midpoint the answer is at
1008 * most its lower neighbour, otherwise nothing is decided */
1009 if (bits == 0)
1010 return false;
1011 NEAREST_ASK(bits - 1, beyond);
1012 if (beyond != 0)
1013 return false;
1014 bits -= 1;
1015 }
1016 if (beyond == 1)
1017 {
1018 /* The answer is above: lo is beyond, hi is not */
1019 lo = bits;
1020 while (true)
1021 {
1022 hi = (lo < max_bits - step) ? lo + step : max_bits;
1023 if (hi == max_bits)
1024 break;
1025 NEAREST_ASK(hi, beyond);
1026 if (beyond == 2)
1027 return false;
1028 if (! beyond)
1029 break;
1030 lo = hi;
1031 if (moves++ > 0)
1032 step *= 2;
1033 }
1034 }
1035 else
1036 {
1037 /* The answer is at or below: hi is not beyond, lo is beyond or below 0 */
1038 hi = bits;
1039 while (true)
1040 {
1041 if (hi == 0)
1042 {
1043 *result = 0;
1044 return true;
1045 }
1046 lo = (hi > step) ? hi - step : 0;
1047 NEAREST_ASK(lo, beyond);
1048 if (beyond == 2)
1049 return false;
1050 if (beyond)
1051 break;
1052 if (lo == 0)
1053 {
1054 *result = 0;
1055 return true;
1056 }
1057 hi = lo;
1058 if (moves++ > 0)
1059 step *= 2;
1060 }
1061 }
1062 while (hi - lo > 1)
1063 {
1064 uint64 mid = lo + (hi - lo) / 2;
1065 NEAREST_ASK(mid, beyond);
1066 if (beyond == 2)
1067 return false;
1068 if (beyond)
1069 lo = mid;
1070 else
1071 hi = mid;
1072 }
1073#undef NEAREST_ASK
1074 *result = hi;
1075 return true;
1076}

1191static pg_attribute_always_inline void
1192ddb_cross3(const DDBound *a, const DDBound *b, DDBound *r)
1193{
1194 bool exact = true;
1195 for (int c = 0; c < 3; c++)
1196 exact = exact && a[c].lo == 0.0 && a[c].e == 0.0 && b[c].lo == 0.0 &&
1197 b[c].e == 0.0;
1198 if (exact)
1199 {
1200 for (int c = 0; c < 3; c++)
1201 {
1202 int i = (c + 1) % 3, j = (c + 2) % 3;
1203 double p1, e1, p2, e2, s, se;
1204 two_product(a[i].hi, b[j].hi, &p1, &e1);
1205 two_product(a[j].hi, b[i].hi, &p2, &e2);
1206 /* Two exact products are equal where their pairs are, the pair of a
1207 * product being unique, and their difference is then an exact zero;
1208 * a product that underflows or overflows is not split exactly, so two
1209 * of them say nothing by being equal: two products past the largest
1210 * double, of one sign, are both an infinity whatever their values */
1211 bool under = (p1 != 0.0 && fabs(p1) < DDB_EXACT_PRODUCT) ||
1212 (p2 != 0.0 && fabs(p2) < DDB_EXACT_PRODUCT) ||
1213 (p1 == 0.0 && a[i].hi != 0.0 && b[j].hi != 0.0) ||
1214 (p2 == 0.0 && a[j].hi != 0.0 && b[i].hi != 0.0);
1215 if (! under && isfinite(p1) && p1 == p2 && e1 == e2)
1216 {
1217 r[c].hi = r[c].lo = r[c].e = 0.0;
1218 continue;
1219 }
1220 two_diff(p1, p2, &s, &se);
1221 se += e1 - e2;
1222 two_sum(s, se, &r[c].hi, &r[c].lo);
1223 r[c].e = DDB_EPS * (fabs(p1) + fabs(p2)) + DDB_TINY;
1224 }
1225 return;
1226 }
1227 for (int c = 0; c < 3; c++)
1228 {
1229 int i = (c + 1) % 3, j = (c + 2) % 3;
1230 r[c] = ddb_sub(ddb_mul(a[i], b[j]), ddb_mul(a[j], b[i]));
1231 }
1232 return;
1233}
