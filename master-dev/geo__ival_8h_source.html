46/*
47 * A value known to lie in [lo, hi]. Each operation rounds its bounds outward
48 * by one step, which covers the rounding to nearest of the operation itself,
49 * so the interval of a computed quantity holds its exact value: a comparison
50 * the intervals decide is the exact comparison.
51 */
52typedef struct
53{
54 double lo;
55 double hi;
56} Ival;
57
58/* fmin and fmax inline, as #float8_min and #float8_max are, but keeping their
59 * rule for NaN, the operand that is not NaN where one is, rather than the
60 * order putting NaN after every number; either of two zeros, which no
61 * comparison tells apart */
62static inline double
63ival_fmin(double a, double b)
64{
65 return (a < b || isnan(b)) ? a : b;
66}
67
68static inline double
69ival_fmax(double a, double b)
70{
71 return (a > b || isnan(b)) ? a : b;
72}

74static inline Ival
75ival(double x)
76{
77 Ival r = {x, x};
78 return r;
79}

81/* Each bound moved outward by at least one step: |x| 2^-52 is no less than
82 * the unit of the last place of x and is exact, 2^-1074 covers the zeros and
83 * the subnormals, and the sum or difference rounds to nearest, which never
84 * brings a value back past the double next to x. An infinity stays one */
85static inline Ival
86ival_out(double lo, double hi)
87{
88 Ival r = {lo - (fabs(lo) * 0x1p-52 + 0x1p-1074),
89 hi + (fabs(hi) * 0x1p-52 + 0x1p-1074)};
90 return r;
91}

93static inline Ival
94ival_add(Ival a, Ival b)
95{
96 return ival_out(a.lo + b.lo, a.hi + b.hi);
97}
98
99static inline Ival
100ival_sub(Ival a, Ival b)
101{
102 return ival_out(a.lo - b.hi, a.hi - b.lo);
103}

105/* The interval of four products or quotients: their extremes, rounded
106 * outward by #ival_out, and the whole line where one is undefined, an
107 * infinity times zero */
108static inline Ival
109ival_extremes(const double p[4])
110{
111 if (isnan(p[0]) || isnan(p[1]) || isnan(p[2]) || isnan(p[3]))
112 {
113 Ival r = {-INFINITY, INFINITY};
114 return r;
115 }
116 double lo = p[0], hi = p[0];
117 for (int i = 1; i < 4; i++)
118 {
119 lo = (p[i] < lo) ? p[i] : lo;
120 hi = (p[i] > hi) ? p[i] : hi;
121 }
122 return ival_out(lo, hi);
123}

125/* The product: where the sign of either interval is known, its extremes are
126 * two of the four products of the bounds, which the signs name; where both
127 * hold zero within them, all four are read (#ival_extremes) */
128static pg_attribute_always_inline Ival
129ival_mul(Ival a, Ival b)
130{
131 double lo, hi;
132 if (a.lo >= 0.0 && b.lo >= 0.0)
133 {
134 lo = a.lo * b.lo; hi = a.hi * b.hi;
135 }
136 else if (a.lo >= 0.0 && b.hi <= 0.0)
137 {
138 lo = a.hi * b.lo; hi = a.lo * b.hi;
139 }
140 else if (a.lo >= 0.0)
141 {
142 lo = a.hi * b.lo; hi = a.hi * b.hi;
143 }
144 else if (a.hi <= 0.0 && b.lo >= 0.0)
145 {
146 lo = a.lo * b.hi; hi = a.hi * b.lo;
147 }
148 else if (a.hi <= 0.0 && b.hi <= 0.0)
149 {
150 lo = a.hi * b.hi; hi = a.lo * b.lo;
151 }
152 else if (a.hi <= 0.0)
153 {
154 lo = a.lo * b.hi; hi = a.lo * b.lo;
155 }
156 else if (b.lo >= 0.0)
157 {
158 lo = a.lo * b.hi; hi = a.hi * b.hi;
159 }
160 else if (b.hi <= 0.0)
161 {
162 lo = a.hi * b.lo; hi = a.lo * b.lo;
163 }
164 else
165 {
166 double p[4] = {a.lo * b.lo, a.lo * b.hi, a.hi * b.lo, a.hi * b.hi};
167 return ival_extremes(p);
168 }
169 if (isnan(lo) || isnan(hi))
170 {
171 Ival r = {-INFINITY, INFINITY};
172 return r;
173 }
174 return ival_out(lo, hi);
175}

177/* A divisor holding zero bounds nothing */
178static pg_attribute_always_inline Ival
179ival_div(Ival a, Ival b)
180{
181 if (b.lo <= 0.0 && b.hi >= 0.0)
182 {
183 Ival r = {-INFINITY, INFINITY};
184 return r;
185 }
186 /* The two quotients bounding it, as the signs choose them (#ival_mul) */
187 double lo, hi;
188 if (b.lo > 0.0)
189 {
190 lo = a.lo / ((a.lo >= 0.0) ? b.hi : b.lo);
191 hi = a.hi / ((a.hi <= 0.0) ? b.hi : b.lo);
192 }
193 else
194 {
195 lo = a.hi / ((a.hi <= 0.0) ? b.lo : b.hi);
196 hi = a.lo / ((a.lo >= 0.0) ? b.lo : b.hi);
197 }
198 if (isnan(lo) || isnan(hi))
199 {
200 Ival r = {-INFINITY, INFINITY};
201 return r;
202 }
203 return ival_out(lo, hi);
204}

220/* The sign an interval decides, or 2 where it holds zero and another sign */
221static inline int
222ival_sign(Ival a)
223{
224 if (a.lo > 0.0)
225 return 1;
226 if (a.hi < 0.0)
227 return -1;
228 if (a.lo == 0.0 && a.hi == 0.0)
229 return 0;
230 return 2;
231}

247ival_hull(Ival a, Ival b)
