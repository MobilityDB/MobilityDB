78/* Analytic distance engine shared with the temporal circular
79 * buffer family: a moving disc (radius may be 0, i.e. a moving point)
80 * against a decomposed geometry, indexed by a Morton bucket hierarchy for
81 * the nearest-approach (knn) kernels and the generic R-tree elsewhere.
82 */
83
89typedef struct
90{
91 double x1, y1, x2, y2;
92 double xmin, ymin, xmax, ymax;
93 bool is_poly;
94 /* Circular-arc edge (from a CIRCULARSTRING): when @p is_arc is true the edge
95 * is the arc of the circle centred at (@p acx, @p acy) with radius @p arad,
96 * from angle @p at0 to @p at1 traversed counterclockwise when @p accw. The
97 * endpoints (x1,y1)/(x2,y2) are the arc start/end; the bounding box already
98 * accounts for the arc bulge (cardinal extremes within the angular span). */
99 bool is_arc;
100 double acx, acy, arad, at0, at1;
101 bool accw;
102} DistEdge;
