92static inline TboxDim
93tbox_kd_dim(int16 flags, int level)
94{
95 bool hasx = MEOS_FLAGS_GET_X(flags), hast = MEOS_FLAGS_GET_T(flags);
96 assert(hasx || hast);
97 if (hasx && hast)
98 return (TboxDim) (level % 4);
99 int half = level % 2;
100 if (hasx)
101 return half ? TBOX_XMAX : TBOX_XMIN;
102 return half ? TBOX_TMAX : TBOX_TMIN;
103}
