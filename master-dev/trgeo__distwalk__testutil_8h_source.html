65static inline double
66tu_segdist(double px, double py, double ax, double ay, double bx, double by)
67{
68 double ux = bx - ax, uy = by - ay, l2 = ux * ux + uy * uy;
69 double s = (l2 > 0.0) ? ((px - ax) * ux + (py - ay) * uy) / l2 : 0.0;
70 s = (s < 0.0) ? 0.0 : ((s > 1.0) ? 1.0 : s);
71 return hypot(px - (ax + s * ux), py - (ay + s * uy));
72}

78static inline bool
79tu_separates(const double *ax, const double *ay, int n, const double *bx,
80 const double *by, int m)
81{
82 for (int i = 0; i < n; i++)
83 {
84 int j = (i + 1) % n;
85 bool all = true;
86 for (int k = 0; k < m && all; k++)
87 if ((ax[j] - ax[i]) * (by[k] - ay[i]) -
88 (ay[j] - ay[i]) * (bx[k] - ax[i]) >= 0.0)
89 all = false;
90 if (all)
91 return true;
92 }
93 return false;
94}

105static inline double
106tu_truedist(const DistMotion *ma, const double *va, int n,
107 const DistMotion *mb, const double *vb, int m, double t)
108{
109 double ax[TU_MAXVERT], ay[TU_MAXVERT], bx[TU_MAXVERT], by[TU_MAXVERT];
110 tu_place(ma, va, n, t, ax, ay);
111 tu_place(mb, vb, m, t, bx, by);
112 if (! tu_separates(ax, ay, n, bx, by, m) &&
113 (m < 2 || ! tu_separates(bx, by, m, ax, ay, n)))
114 return 0.0;
115 double best = INFINITY;
116 for (int i = 0; i < n; i++)
117 for (int k = 0; k < m; k++)
118 {
119 double d = tu_segdist(bx[k], by[k], ax[i], ay[i], ax[(i + 1) % n],
120 ay[(i + 1) % n]);
121 if (d < best)
122 best = d;
123 if (m >= 2)
124 {
125 d = tu_segdist(ax[i], ay[i], bx[k], by[k], bx[(k + 1) % m],
126 by[(k + 1) % m]);
127 if (d < best)
128 best = d;
129 }
130 }
131 return best;
132}

141static inline bool
142tu_refpoly(DistRefPoly *rp, const double *v, int n)
143{
144 if (n == 1)
145 {
146 distrefpoly_set_point(rp);
147 return true;
148 }
149 if (n == 2)
150 {
151 double cx, cy;
152 distrefpoly_set_segment(rp, v[0], v[1], v[2], v[3], &cx, &cy);
153 return true;
154 }
155 POINTARRAY *pa = ptarray_construct_empty(0, 0, (uint32_t) (n + 1));
156 for (int i = 0; i <= n; i++)
157 {
158 POINT4D p = {v[2 * (i % n)], v[2 * (i % n) + 1], 0, 0};
159 ptarray_append_point(pa, &p, LW_TRUE);
160 }
161 POINTARRAY **ringarr = lwalloc(sizeof(POINTARRAY *));
162 ringarr[0] = pa;
163 LWPOLY *poly = lwpoly_construct(0, NULL, 1, ringarr);
164 bool ok = distrefpoly_set(rp, poly);
165 lwpoly_free(poly);
166 return ok;
167}

221static inline void
222tu_check(const DistMotion *ma, const double *va, int n, const DistMotion *mb,
223 const double *vb, int m, const DistEvents *ev, int scan, double level,
224 TuCheck *res)
225{
226 memset(res, 0, sizeof(TuCheck));
227 const DistEvent *e = ev->ev;
228 int ne = ev->count;
229 double emin = INFINITY, bmin = INFINITY;
230 for (int i = 0; i < ne; i++)
231 {
232 double td = tu_truedist(ma, va, n, mb, vb, m, e[i].t);
233 double err = fabs(e[i].dist - td) / (1.0 + td);
234 if (err > res->maxerr)
235 res->maxerr = err;
236 if (e[i].dist < td - 1e-9 * (1.0 + td))
237 res->below++;
238 if (i > 0 && e[i].t <= e[i - 1].t)
239 res->order++;
240 if (e[i].dist < emin)
241 emin = e[i].dist;
242 }
243 int k = 0;
244 for (int j = 0; j <= scan; j++)
245 {
246 double t = (double) j / scan;
247 double td = tu_truedist(ma, va, n, mb, vb, m, t);
248 if (td < bmin)
249 bmin = td;
250 while (k + 2 < ne && e[k + 1].t <= t)
251 k++;
252 double span = e[k + 1].t - e[k].t;
253 double r = (span > 0.0) ? (t - e[k].t) / span : 0.0;
254 double interp = e[k].dist + r * (e[k + 1].dist - e[k].dist);
255 /* Near the level, the true distance cannot decide the side */
256 if (level >= 0.0 && fabs(td - level) > 1e-9 * (1.0 + level) &&
257 (interp <= level) != (td <= level))
258 res->level_side++;
259 if (td > 0.0)
260 continue;
261 res->overlap = true;
262 if (interp > 1e-7)
263 res->nonzero_overlap++;
264 }
265 res->nadgap = (emin - bmin) / (1.0 + bmin);
266 for (int i = 0; i + 1 < ne; i++)
267 {
268 if (e[i].dist != 0.0 || e[i + 1].dist != 0.0)
269 continue;
270 for (int j = 1; j < 16; j++)
271 {
272 double t = e[i].t + (e[i + 1].t - e[i].t) * j / 16.0;
273 if (tu_truedist(ma, va, n, mb, vb, m, t) > 1e-7)
274 {
275 res->zero_apart++;
276 break;
277 }
278 }
279 }
280}
