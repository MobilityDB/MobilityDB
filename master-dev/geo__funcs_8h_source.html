182static inline bool
183geom_ring_is_convex(const POINTARRAY *pa)
184{
185 if (! pa || pa->npoints < 4)
186 return false;
187 uint32_t n = pa->npoints - 1;
188 int sign = 0;
189 for (uint32_t i = 0; i < n; i++)
190 {
191 POINT4D a, b, c;
192 getPoint4d_p(pa, i, &a);
193 getPoint4d_p(pa, (i + 1) % n, &b);
194 getPoint4d_p(pa, (i + 2) % n, &c);
195 double turn = (b.x - a.x) * (c.y - b.y) - (b.y - a.y) * (c.x - b.x);
196 /* The tolerance liblwgeom uses for a coordinate comparison */
197 if (fabs(turn) <= MEOS_GEOM_TOLERANCE)
198 continue;
199 int current = turn > 0.0 ? 1 : -1;
200 if (sign == 0)
201 sign = current;
202 else if (sign != current)
203 return false;
204 }
205 return sign != 0;
206}

319static inline IntersectResult
320linesegm_intersect(double ax, double ay, double rx, double ry,
321 double cx, double cy, double dx, double dy)
322{
323 IntersectResult res = {INTERSECT_NONE, 0, 0};
324 double sx = dx - cx, sy = dy - cy; /* vector CD */
325 /* Where is the start of the second segment relative to the first? */
326 double qpx = cx - ax, qpy = cy - ay;
327
328 /* Are the two segments parallel? */
329 double rxs = rx * sy - ry * sx;
330
331 /* Collinear / parallel */
332 if (fabs(rxs) < MEOS_GEOM_TOLERANCE)
333 {
334 /* Is point C aligned with segment AB? */
335 double qpxr = qpx * ry - qpy * rx;
336 /* If qpxr != 0: parallel, if qpxr == 0: collinear */
337 if (fabs(qpxr) > MEOS_GEOM_TOLERANCE)
338 return res;
339
340 /* Collinear case */
341 double r2 = rx * rx + ry * ry;
342 if (r2 < MEOS_GEOM_TOLERANCE)
343 return res;
344
345 double t0 = (qpx * rx + qpy * ry) / r2;
346 double t1 = t0 + (sx * rx + sy * ry) / r2;
347
348 /* Order t0 < t1 */
349 if (t0 > t1) { double tmp = t0; t0 = t1; t1 = tmp; }
350 /* No intersection */
351 if (t1 < 0 || t0 > 1)
352 return res;
353
354 /* Clamp values */
355 if (t0 < 0) t0 = 0;
356 if (t1 > 1) t1 = 1;
357
358 if (fabs(t1 - t0) < MEOS_GEOM_TOLERANCE)
359 {
360 res.type = INTERSECT_POINT;
361 res.t0 = t0;
362 return res;
363 }
364
365 res.type = INTERSECT_OVERLAP;
366 res.t0 = t0;
367 res.t1 = t1;
368 return res;
369 }
370
371 /* Proper intersection */
372 double t = (qpx * sy - qpy * sx) / rxs;
373 double u = (qpx * ry - qpy * rx) / rxs;
374
375 if (t < -MEOS_GEOM_TOLERANCE || t > 1 + MEOS_GEOM_TOLERANCE ||
376 u < -MEOS_GEOM_TOLERANCE || u > 1 + MEOS_GEOM_TOLERANCE)
377 return res;
378
379 /* Clamp values */
380 if (fabs(t) < MEOS_GEOM_TOLERANCE) t = 0;
381 if (fabs(t - 1) < MEOS_GEOM_TOLERANCE) t = 1;
382
383 res.type = INTERSECT_POINT;
384 res.t0 = t;
385 return res;
386}

399static inline bool
400arc_span_contains(double theta0, double theta1, bool ccw, double phi)
401{
402 double sweep = ccw ?
403 angle_normalize(theta1 - theta0) :
404 angle_normalize(theta0 - theta1);
405 double off = ccw ?
406 angle_normalize(phi - theta0) :
407 angle_normalize(theta0 - phi);
408 return off <= sweep + MEOS_GEOM_TOLERANCE;
409}

414static inline bool
415arc_contains_angle(const Edge *e, double phi)
416{
417 return arc_span_contains(e->theta0, e->theta1, e->ccw, phi);
418}

425static inline void
426arc_set_bbox(Edge *e)
427{
428 double xmin = Min(e->x1, e->x2), xmax = Max(e->x1, e->x2);
429 double ymin = Min(e->y1, e->y2), ymax = Max(e->y1, e->y2);
430 const double ang[4] = {0.0, M_PI_2, M_PI, -M_PI_2};
431 const double ex[4] = {e->cx + e->radius, e->cx, e->cx - e->radius, e->cx};
432 const double ey[4] = {e->cy, e->cy + e->radius, e->cy, e->cy - e->radius};
433 for (int k = 0; k < 4; k++)
434 if (arc_contains_angle(e, ang[k]))
435 {
436 if (ex[k] < xmin) xmin = ex[k];
437 if (ex[k] > xmax) xmax = ex[k];
438 if (ey[k] < ymin) ymin = ey[k];
439 if (ey[k] > ymax) ymax = ey[k];
440 }
441 e->xmin = xmin; e->xmax = xmax; e->ymin = ymin; e->ymax = ymax;
442 return;
443}

448static inline bool
449point_on_arc(double px, double py, const Edge *e)
450{
451 /* The radial distance is read from the coordinates of the point and of the
452 * centre, so what it misses by is a property of their arithmetic and not of
453 * the arc: a few units in the last place of the largest of them. An absolute
454 * band asks a point at a projected 6.4e6 to sit on its own circle to a
455 * thousandth of what that coordinate can express, and the point the engine
456 * itself places on the arc then reads as lying off it. The edge carries the
457 * tolerance its own coordinates call for -- the same quantity
458 * #point_on_segment reads for a straight edge -- and the floor leaves the
459 * band at MEOS_GEOM_TOLERANCE where an edge carries none */
460 double d = hypot(px - e->cx, py - e->cy);
461 if (fabs(d - e->radius) > fmax(e->tol, MEOS_GEOM_TOLERANCE))
462 return false;
463 return arc_contains_angle(e, atan2(py - e->cy, px - e->cx));
464}

479static inline int
480arcsegm_intersect(double ax, double ay, double rx, double ry, const Edge *e,
481 double out[2])
482{
483 double aa = rx * rx + ry * ry;
484 /* Degenerate (zero-length) trajectory segment */
485 if (aa < MEOS_GEOM_TOLERANCE)
486 return 0;
487
488 double wx = ax - e->cx, wy = ay - e->cy;
489 double bb = 2 * (wx * rx + wy * ry);
490 double cc = wx * wx + wy * wy - e->radius * e->radius;
491 double disc = bb * bb - 4 * aa * cc;
492 /* The discriminant is the difference of two quantities of like size, so a
493 * segment touching the circle reaches zero only to their rounding. Reading
494 * it against that rounding rather than against zero keeps a touch one root:
495 * against zero it becomes two roots a square root of the rounding apart,
496 * which is a node pair far enough apart to be taken for two, and the sliver
497 * between them is not a piece of any boundary.
498 * Both terms VANISH where the segment starts on the circle and runs
499 * tangent to it there, which is what the offset of an edge leaving a vertex
500 * does against the round cap about that vertex, so a rounding read from them
501 * alone degenerates to zero and protects nothing. The quantities they are
502 * differences OF do not vanish: cc subtracts two of size radius squared, and
503 * bb is twice a product of size w by the segment direction */
504 double eps = 1e-14 * (fabs(bb * bb) + fabs(4 * aa * cc) +
505 aa * (wx * wx + wy * wy + e->radius * e->radius));
506 if (disc < -eps)
507 return 0;
508 if (disc <= eps)
509 disc = 0;
510
511 double sq = sqrt(disc);
512 double roots[2];
513 int nroots = 0;
514 roots[nroots++] = (-bb - sq) / (2 * aa);
515 /* Distinct second root only when the line is not tangent */
516 if (sq > MEOS_GEOM_TOLERANCE)
517 roots[nroots++] = (-bb + sq) / (2 * aa);
518
519 int n = 0;
520 for (int k = 0; k < nroots; k++)
521 {
522 double t = roots[k];
523 if (t < -MEOS_GEOM_TOLERANCE || t > 1 + MEOS_GEOM_TOLERANCE)
524 continue;
525 if (t < 0) t = 0;
526 if (t > 1) t = 1;
527 double px = ax + t * rx, py = ay + t * ry;
528 if (arc_contains_angle(e, atan2(py - e->cy, px - e->cx)))
529 out[n++] = t;
530 }
531 return n;
532}

544static inline bool
545arcarc_intersect(const Edge *e1, const Edge *e2)
546{
547 double dx = e2->cx - e1->cx, dy = e2->cy - e1->cy;
548 double d = hypot(dx, dy);
549 double r1 = e1->radius, r2 = e2->radius;
550
551 /* Concentric supporting circles */
552 if (d < MEOS_GEOM_TOLERANCE)
553 {
554 if (fabs(r1 - r2) > MEOS_GEOM_TOLERANCE)
555 return false;
556 /* Same circle: the arcs meet iff their spans share an endpoint angle */
557 return arc_contains_angle(e2, e1->theta0) ||
558 arc_contains_angle(e2, e1->theta1) ||
559 arc_contains_angle(e1, e2->theta0) ||
560 arc_contains_angle(e1, e2->theta1);
561 }
562 /* Circles too far apart or one strictly inside the other */
563 if (d > r1 + r2 + MEOS_GEOM_TOLERANCE || d < fabs(r1 - r2) - MEOS_GEOM_TOLERANCE)
564 return false;
565
566 double a = (d * d + r1 * r1 - r2 * r2) / (2 * d);
567 double h2 = r1 * r1 - a * a;
568 if (h2 < 0)
569 h2 = 0;
570 double h = sqrt(h2);
571 double ux = dx / d, uy = dy / d; /* Unit vector from c1 to c2 */
572 double mx = e1->cx + a * ux, my = e1->cy + a * uy; /* Foot on the centre line */
573
574 /* Candidate points m +/- h * perp(u), tested against both arcs' spans */
575 for (int k = 0; k < 2; k++)
576 {
577 double px = mx + (k ? h : -h) * (-uy);
578 double py = my + (k ? h : -h) * ux;
579 if (arc_contains_angle(e1, atan2(py - e1->cy, px - e1->cx)) &&
580 arc_contains_angle(e2, atan2(py - e2->cy, px - e2->cx)))
581 return true;
582 /* A tangency has a single candidate point */
583 if (h < MEOS_GEOM_TOLERANCE)
584 break;
585 }
586 return false;
587}

606static inline double
607coordinate_tolerance(double c1, double c2)
608{
609 return MEOS_GEOM_TOLERANCE + 4.0 * DBL_EPSILON * fmax(fabs(c1), fabs(c2));
610}

642static inline bool
643point_on_segment_within(double px, double py, double x1, double y1, double x2,
644 double y2, double tol)
645{
646 /* Fast bounding-box rejection, which is where all but a few calls end */
647 if ((px < fmin(x1, x2) - tol) || (px > fmax(x1, x2) + tol) ||
648 (py < fmin(y1, y2) - tol) || (py > fmax(y1, y2) + tol))
649 return false;
650
651 /* Vectors AP and AB */
652 double apx = px - x1;
653 double apy = py - y1;
654 double abx = x2 - x1;
655 double aby = y2 - y1;
656
657 /* The tolerance as the area it bounds over a segment of this size */
658 double areatol = tol * (fabs(abx) + fabs(aby));
659
660 /* Collinearity check via cross product */
661 double cross = apx * aby - apy * abx;
662 if (fabs(cross) > areatol)
663 return false;
664
665 /* Projection check via dot product */
666 double dot = apx * abx + apy * aby;
667 if (dot < -areatol)
668 return false;
669
670 /* Check if P lies between A and B */
671 double ab2 = abx * abx + aby * aby;
672 if (dot > ab2 + areatol)
673 return false;
674 return true;
675}

682static inline bool
683point_on_segment(double px, double py, double x1, double y1, double x2,
684 double y2)
685{
686 return point_on_segment_within(px, py, x1, y1, x2, y2,
687 fmax(coordinate_tolerance(x1, x2), coordinate_tolerance(y1, y2)));
688}
