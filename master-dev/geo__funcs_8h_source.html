528static inline bool
529geom_ring_is_convex(const POINTARRAY *pa)
530{
531 if (! pa || pa->npoints < 4)
532 return false;
533 uint32_t n = pa->npoints - 1;
534 int sign = 0;
535 for (uint32_t i = 0; i < n; i++)
536 {
537 POINT4D a, b, c;
538 getPoint4d_p(pa, i, &a);
539 getPoint4d_p(pa, (i + 1) % n, &b);
540 getPoint4d_p(pa, (i + 2) % n, &c);
541 double turn = (b.x - a.x) * (c.y - b.y) - (b.y - a.y) * (c.x - b.x);
542 /* The tolerance liblwgeom uses for a coordinate comparison */
543 if (fabs(turn) <= MEOS_GEOM_TOLERANCE)
544 continue;
545 int current = turn > 0.0 ? 1 : -1;
546 if (sign == 0)
547 sign = current;
548 else if (sign != current)
549 return false;
550 }
551 return sign != 0;
552}

710 double cy, double dx, double dy)
711{
712 double left = (bx - ax) * (dy - cy);
713 double right = (by - ay) * (dx - cx);
714 double det = left - right;
715 double bound = (3.0 + 16.0 * DBL_EPSILON) * DBL_EPSILON *
716 (fabs(left) + fabs(right));
717 if (det > bound)
718 return 1;
719 if (det < - bound)
720 return -1;
721 if (left == 0.0 && right == 0.0)
722 return 0;
723 return cross_product_sign_exact(ax, ay, bx, by, cx, cy, dx, dy);
724}

737static inline int
738dot_product_sign(const POINT3D *p, const POINT3D *q)
739{
740 double xx = p->x * q->x, yy = p->y * q->y, zz = p->z * q->z;
741 double dot = xx + yy + zz;
742 double bound = (2.0 + 16.0 * DBL_EPSILON) * DBL_EPSILON *
743 (fabs(xx) + fabs(yy) + fabs(zz));
744 if (dot > bound)
745 return 1;
746 if (dot < - bound)
747 return -1;
748 return dot_product_sign_exact(p, q);
749}

773static inline int
774point_within_distance_sign(double px, double py, double qx, double qy,
775 double d)
776{
777 double h = qx - px, v = qy - py;
778 double hh = h * h, vv = v * v, dd = d * d;
779 double value = (hh + vv) - dd;
780 /* Three roundings reach each squared difference (the difference, its square,
781 * and the sum) and two reach the square of the distance, so five bound the
782 * whole; the second-order term follows the siblings above, and DBL_MIN
783 * covers a square in the subnormal range */
784 double bound = (5.0 + 32.0 * DBL_EPSILON) * DBL_EPSILON * (hh + vv + dd) +
785 DBL_MIN;
786 if (value > bound)
787 return 1;
788 if (value < - bound)
789 return -1;
790 return point_within_distance_sign_exact(px, py, qx, qy, d);
791}

807static inline int
808triple_product_sign(const POINT3D *p, const POINT3D *q, const POINT3D *r)
809{
810 double qyrz = q->y * r->z, qzry = q->z * r->y;
811 double qzrx = q->z * r->x, qxrz = q->x * r->z;
812 double qxry = q->x * r->y, qyrx = q->y * r->x;
813 double det = p->x * (qyrz - qzry) + p->y * (qzrx - qxrz) +
814 p->z * (qxry - qyrx);
815 double bound = (3.5 + 28.0 * DBL_EPSILON) * DBL_EPSILON *
816 (fabs(p->x) * (fabs(qyrz) + fabs(qzry)) +
817 fabs(p->y) * (fabs(qzrx) + fabs(qxrz)) +
818 fabs(p->z) * (fabs(qxry) + fabs(qyrx)));
819 if (det > bound)
820 return 1;
821 if (det < - bound)
822 return -1;
823 return triple_product_sign_exact(p, q, r);
824}

858static inline IntersectResult
859linesegm_intersect_collinear(double ax, double ay, double bx, double by,
860 double cx, double cy, double dx, double dy)
861{
862 IntersectResult res = {INTERSECT_NONE, 0, 0};
863 if (ax == bx && ay == by)
864 return res;
865 double rx = bx - ax, ry = by - ay;
866 double r2 = rx * rx + ry * ry;
867 bool alongx = fabs(rx) >= fabs(ry);
868 double sgn = ((alongx ? rx : ry) > 0.0) ? 1.0 : -1.0;
869 double pa = sgn * (alongx ? ax : ay), pb = sgn * (alongx ? bx : by);
870 double pc = sgn * (alongx ? cx : cy), pd = sgn * (alongx ? dx : dy);
871 bool cfirst = pc <= pd;
872 double plo = cfirst ? pc : pd, phi = cfirst ? pd : pc;
873 double lox = cfirst ? cx : dx, loy = cfirst ? cy : dy;
874 double hix = cfirst ? dx : cx, hiy = cfirst ? dy : cy;
875 if (plo < pa)
876 {
877 plo = pa; lox = ax; loy = ay;
878 }
879 if (phi > pb)
880 {
881 phi = pb; hix = bx; hiy = by;
882 }
883 if (plo > phi)
884 return res;
885 res.t0 = linesegm_param(ax, ay, bx, by, r2, lox, loy);
886 if (plo == phi)
887 {
888 res.type = INTERSECT_POINT;
889 return res;
890 }
891 res.type = INTERSECT_OVERLAP;
892 res.t1 = linesegm_param(ax, ay, bx, by, r2, hix, hiy);
893 return res;
894}

913static inline IntersectResult
914linesegm_intersect(double ax, double ay, double bx, double by,
915 double cx, double cy, double dx, double dy)
916{
917 IntersectResult res = {INTERSECT_NONE, 0, 0};
918 if (ax == bx && ay == by)
919 return res;
920 double rx = bx - ax, ry = by - ay;
921 double r2 = rx * rx + ry * ry;
922 /* The sides of the line of the first segment the ends of the second lie on.
923 * On one side, the segments cannot meet. Both on it, the second segment lies
924 * on the line of the first. Otherwise the two cross products these are the
925 * signs of differ, and their difference is the cross product of the two
926 * directions, so the directions cross */
927 int oc = cross_product_sign(ax, ay, bx, by, ax, ay, cx, cy);
928 int od = cross_product_sign(ax, ay, bx, by, ax, ay, dx, dy);
929 if ((oc > 0 && od > 0) || (oc < 0 && od < 0))
930 return res;
931
932 if (oc == 0 && od == 0)
933 return linesegm_intersect_collinear(ax, ay, bx, by, cx, cy, dx, dy);
934
935 /* Crossing directions: the segments meet where the first does not lie
936 * wholly on one side of the line of the second either */
937 int oa = cross_product_sign(cx, cy, dx, dy, cx, cy, ax, ay);
938 int ob = cross_product_sign(cx, cy, dx, dy, cx, cy, bx, by);
939 if ((oa > 0 && ob > 0) || (oa < 0 && ob < 0))
940 return res;
941 res.type = INTERSECT_POINT;
942 if (oa == 0)
943 res.t0 = 0.0;
944 else if (ob == 0)
945 res.t0 = 1.0;
946 else if (oc == 0)
947 res.t0 = linesegm_param(ax, ay, bx, by, r2, cx, cy);
948 else if (od == 0)
949 res.t0 = linesegm_param(ax, ay, bx, by, r2, dx, dy);
950 else
951 {
952 /* A crossing interior to both, whose position is a quotient */
953 double sx = dx - cx, sy = dy - cy;
954 double t = ((cx - ax) * sy - (cy - ay) * sx) / (rx * sy - ry * sx);
955 res.t0 = (t < 0.0) ? 0.0 : ((t > 1.0) ? 1.0 : t);
956 }
957 return res;
958}

1050static inline void
1051arc_turn_read(const Edge *e, ArcTurn *t)
1052{
1053 /* Where each end stands from the centre along each axis, exactly */
1054 int ux = (e->x1 > e->cx) - (e->x1 < e->cx);
1055 int uy = (e->y1 > e->cy) - (e->y1 < e->cy);
1056 int vx = (e->x2 > e->cx) - (e->x2 < e->cx);
1057 int vy = (e->y2 > e->cy) - (e->y2 < e->cy);
1058 int turn = cross_product_sign(e->cx, e->cy, e->x1, e->y1, e->cx, e->cy,
1059 e->x2, e->y2);
1060 if (e->ccw)
1061 {
1062 t->ux = ux; t->uy = uy; t->vx = vx; t->vy = vy; t->turn = turn;
1063 }
1064 else
1065 {
1066 /* An arc traversed clockwise from its start to its end covers what the
1067 * same arc traversed the other way covers from its end to its start */
1068 t->ux = vx; t->uy = vy; t->vx = ux; t->vy = uy; t->turn = - turn;
1069 }
1070 return;
1071}
1072
1078static inline bool
1079arc_turn_reaches(const ArcTurn *t, int wx, int wy)
1080{
1081 /* The direction seen from each end: the cross product of the start with it,
1082 * and of it with the end, each keeping the one component that survives */
1083 int su = wy ? wy * t->ux : - wx * t->uy;
1084 int sv = wx ? wx * t->vy : - wy * t->vx;
1085 int turn = t->turn;
1086 if (turn > 0)
1087 /* The arc turns through less than a half circle, so it reaches the
1088 * direction only by that lying after its start and before its end */
1089 return su >= 0 && sv >= 0;
1090 if (turn < 0)
1091 /* Through more than a half circle, where those two conditions hold
1092 * together outside the arc rather than inside it */
1093 return su >= 0 || sv >= 0;
1094 /* The ends stand on one line through the centre: a half circle where they
1095 * stand on opposite sides of it, and no extent where they stand together */
1096 if (t->ux == - t->vx && t->uy == - t->vy)
1097 return su >= 0;
1098 return su == 0 && sv == 0;
1099}

1132static inline bool
1133point_on_arc(double px, double py, const Edge *e)
1134{
1135 /* The radial distance is read from the coordinates of the point and of the
1136 * centre, so what it misses by is a property of their arithmetic and not of
1137 * the arc: a few units in the last place of the largest of them. An absolute
1138 * band asks a point at a projected 6.4e6 to sit on its own circle to a
1139 * thousandth of what that coordinate can express, and the point the engine
1140 * itself places on the arc then reads as lying off it. The edge carries the
1141 * tolerance its own coordinates call for -- the same quantity
1142 * #point_on_segment reads for a straight edge */
1143 double d = hypot(px - e->cx, py - e->cy);
1144 if (fabs(d - e->radius) > e->tol)
1145 return false;
1146 return arc_contains_angle(e, atan2(py - e->cy, px - e->cx));

1171static inline int
1172arcsegm_end_crossing(double ax, double ay, double rx, double ry, const Edge *e,
1173 bool at_start)
1174{
1175 double sx = at_start ? e->x1 : e->x2, sy = at_start ? e->y1 : e->y2;
1176 int side = cross_product_sign(ax, ay, ax + rx, ay + ry, ax, ay, sx, sy);
1177 if (side == 0)
1178 return -1;
1179 /* The tangent in the sense the arc is traversed, turned around at its end,
1180 * where the arc is left by walking it backwards */
1181 double tx = - (sy - e->cy), ty = sx - e->cx;
1182 if (! e->ccw)
1183 {
1184 tx = - tx; ty = - ty;
1185 }
1186 if (! at_start)
1187 {
1188 tx = - tx; ty = - ty;
1189 }
1190 double turn = rx * ty - ry * tx;
1191 if (fabs(turn) <= 4.0 * DBL_EPSILON * (fabs(rx * ty) + fabs(ry * tx)))
1192 return -1;
1193 return ((turn > 0.0) != (side > 0)) ? 1 : 0;
1194}

1209static inline int
1210arcsegm_intersect(double ax, double ay, double rx, double ry, const Edge *e,
1211 double out[2])
1212{
1213 /* A segment of no length is one whose vector is zero. A bound on the square
1214 * of its length in length units drops every segment shorter than the square
1215 * root of that bound, and which segments those are depends on the scale of
1216 * the coordinates */
1217 if (rx == 0.0 && ry == 0.0)
1218 return 0;
1219 double aa = rx * rx + ry * ry;
1220
1221 double wx = ax - e->cx, wy = ay - e->cy;
1222 double bb = 2 * (wx * rx + wy * ry);
1223 double cc = wx * wx + wy * wy - e->radius * e->radius;
1224 double disc = bb * bb - 4 * aa * cc;
1225 /* The discriminant is the difference of two quantities of like size, so a
1226 * segment touching the circle reaches zero only to their rounding. Reading
1227 * it against that rounding rather than against zero keeps a touch one root:
1228 * against zero it becomes two roots a square root of the rounding apart,
1229 * which is a node pair far enough apart to be taken for two, and the sliver
1230 * between them is not a piece of any boundary.
1231 * Both terms VANISH where the segment starts on the circle and runs
1232 * tangent to it there, which is what the offset of an edge leaving a vertex
1233 * does against the round cap about that vertex, so a rounding read from them
1234 * alone degenerates to zero and protects nothing. The quantities they are
1235 * differences OF do not vanish: cc subtracts two of size radius squared, and
1236 * bb is twice a product of size w by the segment direction */
1237 double eps = 1e-14 * (fabs(bb * bb) + fabs(4 * aa * cc) +
1238 aa * (wx * wx + wy * wy + e->radius * e->radius));
1239 if (disc < -eps)
1240 return 0;
1241 if (disc <= eps)
1242 disc = 0;
1243
1244 double sq = sqrt(disc);
1245 double roots[2];
1246 int nroots = 0;
1247 roots[nroots++] = (-bb - sq) / (2 * aa);
1248 /* Distinct second root only when the line is not tangent, which the
1249 * discriminant read against its rounding above already decides */
1250 if (disc > 0.0)
1251 roots[nroots++] = (-bb + sq) / (2 * aa);
1252
1253 int n = 0;
1254 for (int k = 0; k < nroots; k++)
1255 {
1256 double t = roots[k];
1257 if (t < -MEOS_GEOM_TOLERANCE || t > 1 + MEOS_GEOM_TOLERANCE)
1258 continue;
1259 if (t < 0) t = 0;
1260 if (t > 1) t = 1;
1261 double px = ax + t * rx, py = ay + t * ry;
1262 /* A crossing within the rounding of an end of the arc is decided by the
1263 * side that end stands on, where that decides it */
1264 int at_end = -1;
1265 if (fabs(px - e->x1) <= e->tol && fabs(py - e->y1) <= e->tol)
1266 at_end = arcsegm_end_crossing(ax, ay, rx, ry, e, true);
1267 else if (fabs(px - e->x2) <= e->tol && fabs(py - e->y2) <= e->tol)
1268 at_end = arcsegm_end_crossing(ax, ay, rx, ry, e, false);
1269 if (at_end >= 0 ? at_end == 1 :
1270 arc_contains_angle(e, atan2(py - e->cy, px - e->cx)))
1271 out[n++] = t;
1272 }
1273 return n;
1274}

1284static inline double
1285arc_point_parameter(const Edge *e, double x, double y)
1286{
1287 double phi = atan2(y - e->cy, x - e->cx);
1288 double sweep = arc_span_sweep(e->theta0, e->theta1, e->ccw);
1289 if (sweep < MEOS_GEOM_TOLERANCE)
1290 return 0.0;
1291 double off = arc_span_offset(e->theta0, e->ccw, phi);
1292 double t = off / sweep;
1293 /* A point the arithmetic puts just OUTSIDE the span belongs to whichever end
1294 * it stands nearer, and one a rounding step BEFORE the start has turned
1295 * nearly the whole circle rather than none of it */
1296 if (t > 1.0 && (2 * M_PI - off) < (off - sweep))
1297 return 0.0;
1298 return (t < 0.0) ? 0.0 : ((t > 1.0) ? 1.0 : t);
1299}

1322static inline bool
1323arcsegm_cross(double ax, double ay, double rx, double ry, const Edge *e)
1324{
1325 /* A segment of no length, read as #arcsegm_intersect reads it */
1326 if (rx == 0.0 && ry == 0.0)
1327 return false;
1328 double aa = rx * rx + ry * ry;
1329
1330 double wx = ax - e->cx, wy = ay - e->cy;
1331 double bb = 2 * (wx * rx + wy * ry);
1332 double cc = wx * wx + wy * wy - e->radius * e->radius;
1333 double disc = bb * bb - 4 * aa * cc;
1334 /* The rounding the discriminant is read against, in the form
1335 * #arcsegm_intersect states and for the reason it gives there */
1336 double eps = 1e-14 * (fabs(bb * bb) + fabs(4 * aa * cc) +
1337 aa * (wx * wx + wy * wy + e->radius * e->radius));
1338 /* A tangent line reaches the circle without passing through it */
1339 if (disc <= eps)
1340 return false;
1341
1342 double sq = sqrt(disc);
1343 for (int k = 0; k < 2; k++)
1344 {
1345 double t = (k == 0 ? -bb - sq : -bb + sq) / (2 * aa);
1346 /* Strictly interior to the segment. A crossing at an endpoint is a touch,
1347 * which the vertex routes answer and which the straight predicate refuses
1348 * for the same reason */
1349 if (t <= MEOS_GEOM_TOLERANCE || t >= 1 - MEOS_GEOM_TOLERANCE)
1350 continue;
1351 double px = ax + t * rx, py = ay + t * ry;
1352 if (! arc_contains_angle(e, atan2(py - e->cy, px - e->cx)))
1353 continue;
1354 /* Strictly interior to the arc, for the same reason */
1355 double u = arc_point_parameter(e, px, py);
1356 if (u <= MEOS_GEOM_TOLERANCE || u >= 1 - MEOS_GEOM_TOLERANCE)
1357 continue;
1358 return true;
1359 }
1360 return false;
1361}

1376arc_half_chord_rounding(double d, double r1, double r2, double aa)
1377{
1378 return 4.0 * DBL_EPSILON * (r1 * r1 + aa * aa +

1401static inline bool
1402arcarc_cross(const Edge *a, const Edge *b)
1403{
1404 /* One circle: a shared stretch or a common endpoint, never a crossing */
1405 if (arc_same_circle(a, b))
1406 return false;
1407 double dx = b->cx - a->cx, dy = b->cy - a->cy;
1408 double d = hypot(dx, dy);
1409 /* Two circles about one centre never meet */
1410 if (d == 0.0)
1411 return false;
1412 double aa = (d * d + a->radius * a->radius - b->radius * b->radius) /
1413 (2.0 * d);
1414 double h2 = a->radius * a->radius - aa * aa;
1415 /* Separate and nested circles never meet, and tangent ones touch at one
1416 * point without passing through */
1417 if (h2 <= arc_half_chord_rounding(d, a->radius, b->radius, aa))
1418 return false;
1419 double h = sqrt(h2);
1420
1421 /* HOW FAR THE CROSSING POINT ITSELF IS RESOLVED, and why a parameter margin
1422 * cannot stand in for it. The half-chord comes from `r^2 - aa^2`, two
1423 * quantities of size radius squared differenced down to the chord: at a
1424 * radius of 2e5 each is 4e10, whose last representable bit is 8e-6, so the
1425 * point slides along the chord by about that much however exactly the input
1426 * names it. A test asking only that the parameter be some tiny amount
1427 * inside `[0, 1]` therefore reads an ENDPOINT as an interior crossing --
1428 * measured, two arcs meeting exactly at their two shared endpoints report a
1429 * crossing 3.8e-6 away from one of them. The distance below is that
1430 * rounding written out, in the same form #arcsegm_intersect states its own,
1431 * so a candidate nearer an endpoint than the arithmetic can resolve is
1432 * refused rather than believed */
1433 double resolution = DBL_EPSILON *
1434 (a->radius * a->radius + aa * aa) / (2.0 * h);
1435 double ux = dx / d, uy = dy / d;
1436 double mx = a->cx + aa * ux, my = a->cy + aa * uy;
1437 for (int k = 0; k < 2; k++)
1438 {
1439 double px = mx + (k ? h : -h) * (-uy);
1440 double py = my + (k ? h : -h) * ux;
1441 /* On both arcs at all */
1442 if (! arc_contains_angle(a, atan2(py - a->cy, px - a->cx)) ||
1443 ! arc_contains_angle(b, atan2(py - b->cy, px - b->cx)))
1444 continue;
1445 /* Clear of every endpoint by more than the point is resolved to. A
1446 * meeting at an endpoint is a touch, which the vertex routes answer and
1447 * which the straight predicate refuses for the same reason */
1448 if (hypot(px - a->x1, py - a->y1) <= resolution ||
1449 hypot(px - a->x2, py - a->y2) <= resolution ||
1450 hypot(px - b->x1, py - b->y1) <= resolution ||
1451 hypot(px - b->x2, py - b->y2) <= resolution)
1452 continue;
1453 double ua = arc_point_parameter(a, px, py);
1454 if (ua <= MEOS_GEOM_TOLERANCE || ua >= 1 - MEOS_GEOM_TOLERANCE)
1455 continue;
1456 double ub = arc_point_parameter(b, px, py);
1457 if (ub <= MEOS_GEOM_TOLERANCE || ub >= 1 - MEOS_GEOM_TOLERANCE)
1458 continue;
1459 return true;
1460 }
1461 return false;
1462}

1489arcarc_intersect_points(const Edge *e1, const Edge *e2, double ix[2],

1492 double dx = e2->cx - e1->cx, dy = e2->cy - e1->cy;
1493 double d = hypot(dx, dy);
1494 double r1 = e1->radius, r2 = e2->radius;
1495 if (same_circle)
1496 *same_circle = false;
1497
1498 /* Concentric supporting circles */
1499 if (d < MEOS_GEOM_TOLERANCE)
1500 {
1501 if (fabs(r1 - r2) > MEOS_GEOM_TOLERANCE)
1502 return 0;
1503 /* Same circle: the arcs run along one another wherever their spans do,
1504 * which is a shared stretch rather than a pair of points */
1505 if (same_circle)
1506 *same_circle = arc_contains_angle(e2, e1->theta0) ||
1507 arc_contains_angle(e2, e1->theta1) ||
1508 arc_contains_angle(e1, e2->theta0) ||
1509 arc_contains_angle(e1, e2->theta1);
1510 return 0;
1511 }
1512 /* Circles too far apart or one strictly inside the other */
1513 if (d > r1 + r2 + MEOS_GEOM_TOLERANCE || d < fabs(r1 - r2) - MEOS_GEOM_TOLERANCE)
1514 return 0;
1515
1516 double a = (d * d + r1 * r1 - r2 * r2) / (2 * d);
1517 double h2 = r1 * r1 - a * a;
1518 if (h2 < 0)
1519 h2 = 0;
1520 double h = sqrt(h2);
1521 double ux = dx / d, uy = dy / d; /* Unit vector from c1 to c2 */
1522 double mx = e1->cx + a * ux, my = e1->cy + a * uy; /* Foot on the centre line */
1523
1524 /* Candidate points m +/- h * perp(u), kept when both spans hold them */
1525 int n = 0;
1526 for (int k = 0; k < 2; k++)
1527 {
1528 double px = mx + (k ? h : -h) * (-uy);
1529 double py = my + (k ? h : -h) * ux;
1530 if (arc_contains_angle(e1, atan2(py - e1->cy, px - e1->cx)) &&
1531 arc_contains_angle(e2, atan2(py - e2->cy, px - e2->cx)))
1532 {
1533 ix[n] = px; iy[n] = py; n++;
1534 }
1535 /* A tangency has a single candidate point */
1536 if (h < MEOS_GEOM_TOLERANCE)
1537 break;
1538 }
1539 return n;
1540}

1628 double y2, double tol)
1629{
1630 /* Fast bounding-box rejection, which is where all but a few calls end */
1631 if ((px < Min(x1, x2) - tol) || (px > Max(x1, x2) + tol) ||
1632 (py < Min(y1, y2) - tol) || (py > Max(y1, y2) + tol))
1633 return false;
1634
1635 /* Vectors AP and AB */
1636 double apx = px - x1;
1637 double apy = py - y1;
1638 double abx = x2 - x1;
1639 double aby = y2 - y1;
1640
1641 /* The tolerance as the area it bounds over a segment of this size */
1642 double areatol = tol * (fabs(abx) + fabs(aby));
1643
1644 /* Collinearity check via cross product */
1645 double cross = apx * aby - apy * abx;
1646 if (fabs(cross) > areatol)
1647 return false;
1648
1649 /* Projection check via dot product */
1650 double dot = apx * abx + apy * aby;
1651 if (dot < -areatol)
1652 return false;
1653
1654 /* Check if P lies between A and B */
1655 double ab2 = abx * abx + aby * aby;
1656 if (dot > ab2 + areatol)
1657 return false;
1658 return true;
1659}
