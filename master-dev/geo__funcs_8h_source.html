613static inline bool
614geom_ring_is_convex(const POINTARRAY *pa)
615{
616 if (! pa || pa->npoints < 4)
617 return false;
618 uint32_t n = pa->npoints - 1;
619 int sign = 0;
620 for (uint32_t i = 0; i < n; i++)
621 {
622 POINT4D a, b, c;
623 getPoint4d_p(pa, i, &a);
624 getPoint4d_p(pa, (i + 1) % n, &b);
625 getPoint4d_p(pa, (i + 2) % n, &c);
626 double turn = (b.x - a.x) * (c.y - b.y) - (b.y - a.y) * (c.x - b.x);
627 /* The tolerance liblwgeom uses for a coordinate comparison */
628 if (fabs(turn) <= MEOS_GEOM_TOLERANCE)
629 continue;
630 int current = turn > 0.0 ? 1 : -1;
631 if (sign == 0)
632 sign = current;
633 else if (sign != current)
634 return false;
635 }
636 return sign != 0;
637}

821static inline int
822cross_product_sign(double ax, double ay, double bx, double by, double cx,
823 double cy, double dx, double dy)
824{
825 double l1 = bx - ax, l2 = dy - cy, r1 = by - ay, r2 = dx - cx;
826 double left = l1 * l2, right = r1 * r2;
827 double mag = fabs(left) + fabs(right);
828 double bound = (3.0 + 16.0 * DBL_EPSILON) * DBL_EPSILON * mag;
829 if (mag >= 0x1p-969)
830 {
831 double det = left - right;
832 if (det > bound)
833 return 1;
834 if (det < - bound)
835 return -1;
836 }
837 if ((l1 == 0.0 || l2 == 0.0) && (r1 == 0.0 || r2 == 0.0))
838 return 0;
839 return cross_product_sign_exact(ax, ay, bx, by, cx, cy, dx, dy);
840}

880static inline int
881point_within_distance_sign(double px, double py, double qx, double qy,
882 double d)
883{
884 double h = qx - px, v = qy - py;
885 double hh = h * h, vv = v * v, dd = d * d;
886 double value = (hh + vv) - dd;
887 /* Three roundings reach each squared difference (the difference, its square,
888 * and the sum) and two reach the square of the distance, so five bound the
889 * whole; the second-order term follows the siblings above, and DBL_MIN
890 * covers a square in the subnormal range */
891 double bound = (5.0 + 32.0 * DBL_EPSILON) * DBL_EPSILON * (hh + vv + dd) +
892 DBL_MIN;
893 if (value > bound)
894 return 1;
895 if (value < - bound)
896 return -1;
897 return point_within_distance_sign_exact(px, py, qx, qy, d);
898}

952static inline IntersectResult
953linesegm_intersect_collinear(double ax, double ay, double bx, double by,
954 double cx, double cy, double dx, double dy)
955{
956 IntersectResult res = {INTERSECT_NONE, 0, 0};
957 if (ax == bx && ay == by)
958 return res;
959 double rx = bx - ax, ry = by - ay;
960 double r2 = rx * rx + ry * ry;
961 bool alongx = fabs(rx) >= fabs(ry);
962 double sgn = ((alongx ? rx : ry) > 0.0) ? 1.0 : -1.0;
963 double pa = sgn * (alongx ? ax : ay), pb = sgn * (alongx ? bx : by);
964 double pc = sgn * (alongx ? cx : cy), pd = sgn * (alongx ? dx : dy);
965 bool cfirst = pc <= pd;
966 double plo = cfirst ? pc : pd, phi = cfirst ? pd : pc;
967 double lox = cfirst ? cx : dx, loy = cfirst ? cy : dy;
968 double hix = cfirst ? dx : cx, hiy = cfirst ? dy : cy;
969 if (plo < pa)
970 {
971 plo = pa; lox = ax; loy = ay;
972 }
973 if (phi > pb)
974 {
975 phi = pb; hix = bx; hiy = by;
976 }
977 if (plo > phi)
978 return res;
979 res.t0 = linesegm_param(ax, ay, bx, by, r2, lox, loy);
980 if (plo == phi)
981 {
982 res.type = INTERSECT_POINT;
983 return res;
984 }
985 res.type = INTERSECT_OVERLAP;
986 res.t1 = linesegm_param(ax, ay, bx, by, r2, hix, hiy);
987 return res;
988}

1007static inline IntersectResult
1008linesegm_intersect(double ax, double ay, double bx, double by,
1009 double cx, double cy, double dx, double dy)
1010{
1011 IntersectResult res = {INTERSECT_NONE, 0, 0};
1012 if (ax == bx && ay == by)
1013 return res;
1014 double rx = bx - ax, ry = by - ay;
1015 double r2 = rx * rx + ry * ry;
1016 /* The sides of the line of the first segment the ends of the second lie on.
1017 * On one side, the segments cannot meet. Both on it, the second segment lies
1018 * on the line of the first. Otherwise the two cross products these are the
1019 * signs of differ, and their difference is the cross product of the two
1020 * directions, so the directions cross */
1021 int oc = cross_product_sign(ax, ay, bx, by, ax, ay, cx, cy);
1022 int od = cross_product_sign(ax, ay, bx, by, ax, ay, dx, dy);
1023 if ((oc > 0 && od > 0) || (oc < 0 && od < 0))
1024 return res;
1025
1026 if (oc == 0 && od == 0)
1027 return linesegm_intersect_collinear(ax, ay, bx, by, cx, cy, dx, dy);
1028
1029 /* Crossing directions: the segments meet where the first does not lie
1030 * wholly on one side of the line of the second either */
1031 int oa = cross_product_sign(cx, cy, dx, dy, cx, cy, ax, ay);
1032 int ob = cross_product_sign(cx, cy, dx, dy, cx, cy, bx, by);
1033 if ((oa > 0 && ob > 0) || (oa < 0 && ob < 0))
1034 return res;
1035 res.type = INTERSECT_POINT;
1036 if (oa == 0)
1037 res.t0 = 0.0;
1038 else if (ob == 0)
1039 res.t0 = 1.0;
1040 else if (oc == 0)
1041 res.t0 = linesegm_param(ax, ay, bx, by, r2, cx, cy);
1042 else if (od == 0)
1043 res.t0 = linesegm_param(ax, ay, bx, by, r2, dx, dy);
1044 else
1045 {
1046 /* A crossing interior to both, whose position is a quotient */
1047 double sx = dx - cx, sy = dy - cy;
1048 double t = ((cx - ax) * sy - (cy - ay) * sx) / (rx * sy - ry * sx);
1049 res.t0 = (t < 0.0) ? 0.0 : ((t > 1.0) ? 1.0 : t);
1050 }
1051 return res;
1052}

1144static inline void
1145arc_turn_read(const Edge *e, ArcTurn *t)
1146{
1147 /* Where each end stands from the centre along each axis, exactly */
1148 int ux = (e->x1 > e->cx) - (e->x1 < e->cx);
1149 int uy = (e->y1 > e->cy) - (e->y1 < e->cy);
1150 int vx = (e->x2 > e->cx) - (e->x2 < e->cx);
1151 int vy = (e->y2 > e->cy) - (e->y2 < e->cy);
1152 int turn = cross_product_sign(e->cx, e->cy, e->x1, e->y1, e->cx, e->cy,
1153 e->x2, e->y2);
1154 if (e->ccw)
1155 {
1156 t->ux = ux; t->uy = uy; t->vx = vx; t->vy = vy; t->turn = turn;
1157 }
1158 else
1159 {
1160 /* An arc traversed clockwise from its start to its end covers what the
1161 * same arc traversed the other way covers from its end to its start */
1162 t->ux = vx; t->uy = vy; t->vx = ux; t->vy = uy; t->turn = - turn;
1163 }
1164 return;
1165}

1172static inline bool
1173arc_turn_reaches(const ArcTurn *t, int wx, int wy)
1174{
1175 /* The direction seen from each end: the cross product of the start with it,
1176 * and of it with the end, each keeping the one component that survives */
1177 int su = wy ? wy * t->ux : - wx * t->uy;
1178 int sv = wx ? wx * t->vy : - wy * t->vx;
1179 int turn = t->turn;
1180 if (turn > 0)
1181 /* The arc turns through less than a half circle, so it reaches the
1182 * direction only by that lying after its start and before its end */
1183 return su >= 0 && sv >= 0;
1184 if (turn < 0)
1185 /* Through more than a half circle, where those two conditions hold
1186 * together outside the arc rather than inside it */
1187 return su >= 0 || sv >= 0;
1188 /* The ends stand on one line through the centre: a half circle where they
1189 * stand on opposite sides of it, and no extent where they stand together */
1190 if (t->ux == - t->vx && t->uy == - t->vy)
1191 return su >= 0;
1192 return su == 0 && sv == 0;
1193}

1226static inline bool
1227point_on_arc(double px, double py, const Edge *e)
1228{

1230 * centre, so what it misses by is a property of their arithmetic and not of
1231 * the arc: a few units in the last place of the largest of them. An absolute
1232 * band asks a point at a projected 6.4e6 to sit on its own circle to a
1233 * thousandth of what that coordinate can express, and the point the engine

1235 * tolerance its own coordinates call for -- the same quantity
1236 * #point_on_segment reads for a straight edge */

1239 return false;

1265static inline int
1266arcsegm_end_crossing(double ax, double ay, double rx, double ry, const Edge *e,
1267 bool at_start)
1268{
1269 double sx = at_start ? e->x1 : e->x2, sy = at_start ? e->y1 : e->y2;
1270 int side = cross_product_sign(ax, ay, ax + rx, ay + ry, ax, ay, sx, sy);
1271 if (side == 0)
1272 return -1;
1273 /* The tangent in the sense the arc is traversed, turned around at its end,
1274 * where the arc is left by walking it backwards */
1275 double tx = - (sy - e->cy), ty = sx - e->cx;
1276 if (! e->ccw)
1277 {
1278 tx = - tx; ty = - ty;
1279 }
1280 if (! at_start)
1281 {
1282 tx = - tx; ty = - ty;
1283 }
1284 double turn = rx * ty - ry * tx;
1285 if (fabs(turn) <= 4.0 * DBL_EPSILON * (fabs(rx * ty) + fabs(ry * tx)))
1286 return -1;
1287 return ((turn > 0.0) != (side > 0)) ? 1 : 0;

1303static inline int
1304arcsegm_intersect(double ax, double ay, double rx, double ry, const Edge *e,
1305 double out[2])
1306{
1307 /* A segment of no length is one whose vector is zero. A bound on the square
1308 * of its length in length units drops every segment shorter than the square
1309 * root of that bound, and which segments those are depends on the scale of
1310 * the coordinates */
1311 if (rx == 0.0 && ry == 0.0)
1312 return 0;
1313 double aa = rx * rx + ry * ry;
1314
1315 double wx = ax - e->cx, wy = ay - e->cy;
1316 double bb = 2 * (wx * rx + wy * ry);
1317 double cc = wx * wx + wy * wy - e->radius * e->radius;
1318 double disc = bb * bb - 4 * aa * cc;
1319 /* The discriminant is the difference of two quantities of like size, so a
1320 * segment touching the circle reaches zero only to their rounding. Reading
1321 * it against that rounding rather than against zero keeps a touch one root:
1322 * against zero it becomes two roots a square root of the rounding apart,
1323 * which is a node pair far enough apart to be taken for two, and the sliver
1324 * between them is not a piece of any boundary.
1325 * Both terms VANISH where the segment starts on the circle and runs
1326 * tangent to it there, which is what the offset of an edge leaving a vertex
1327 * does against the round cap about that vertex, so a rounding read from them
1328 * alone degenerates to zero and protects nothing. The quantities they are
1329 * differences OF do not vanish: cc subtracts two of size radius squared, and
1330 * bb is twice a product of size w by the segment direction */
1331 double eps = 1e-14 * (fabs(bb * bb) + fabs(4 * aa * cc) +
1332 aa * (wx * wx + wy * wy + e->radius * e->radius));
1333 if (disc < -eps)
1334 return 0;
1335 if (disc <= eps)
1336 disc = 0;
1337
1338 double sq = sqrt(disc);
1339 double roots[2];
1340 int nroots = 0;
1341 roots[nroots++] = (-bb - sq) / (2 * aa);
1342 /* Distinct second root only when the line is not tangent, which the
1343 * discriminant read against its rounding above already decides */
1344 if (disc > 0.0)
1345 roots[nroots++] = (-bb + sq) / (2 * aa);
1346
1347 int n = 0;
1348 for (int k = 0; k < nroots; k++)
1349 {
1350 double t = roots[k];
1351 if (t < -MEOS_GEOM_TOLERANCE || t > 1 + MEOS_GEOM_TOLERANCE)
1352 continue;
1353 if (t < 0) t = 0;
1354 if (t > 1) t = 1;
1355 double px = ax + t * rx, py = ay + t * ry;
1356 /* A crossing within the rounding of an end of the arc is decided by the
1357 * side that end stands on, where that decides it */
1358 int at_end = -1;
1359 if (fabs(px - e->x1) <= e->tol && fabs(py - e->y1) <= e->tol)
1360 at_end = arcsegm_end_crossing(ax, ay, rx, ry, e, true);
1361 else if (fabs(px - e->x2) <= e->tol && fabs(py - e->y2) <= e->tol)
1362 at_end = arcsegm_end_crossing(ax, ay, rx, ry, e, false);
1363 if (at_end >= 0 ? at_end == 1 :
1364 arc_contains_angle(e, atan2(py - e->cy, px - e->cx)))
1365 out[n++] = t;
1366 }
1367 return n;
1368}

1498static inline int
1499arcarc_intersect_points(const Edge *e1, const Edge *e2, double ix[2],
1500 double iy[2], bool *same_circle)
1501{
1502 double dx = e2->cx - e1->cx, dy = e2->cy - e1->cy;
1503 double d = hypot(dx, dy);
1504 double r1 = e1->radius, r2 = e2->radius;
1505 if (same_circle)
1506 *same_circle = false;
1507
1508 /* Concentric supporting circles */
1509 if (d < MEOS_GEOM_TOLERANCE)
1510 {
1511 if (fabs(r1 - r2) > MEOS_GEOM_TOLERANCE)
1512 return 0;
1513 /* Same circle: the arcs run along one another wherever their spans do,
1514 * which is a shared stretch rather than a pair of points */
1515 if (same_circle)
1516 *same_circle = arc_contains_angle(e2, e1->theta0) ||
1517 arc_contains_angle(e2, e1->theta1) ||
1518 arc_contains_angle(e1, e2->theta0) ||
1519 arc_contains_angle(e1, e2->theta1);
1520 return 0;
1521 }
1522 /* Circles too far apart or one strictly inside the other */
1523 if (d > r1 + r2 + MEOS_GEOM_TOLERANCE || d < fabs(r1 - r2) - MEOS_GEOM_TOLERANCE)
1524 return 0;
1525
1526 double a = (d * d + r1 * r1 - r2 * r2) / (2 * d);
1527 double h2 = r1 * r1 - a * a;
1528 if (h2 < 0)
1529 h2 = 0;
1530 double h = sqrt(h2);
1531 double ux = dx / d, uy = dy / d; /* Unit vector from c1 to c2 */
1532 double mx = e1->cx + a * ux, my = e1->cy + a * uy; /* Foot on the centre line */
1533
1534 /* Candidate points m +/- h * perp(u), kept when both spans hold them */
1535 int n = 0;
1536 for (int k = 0; k < 2; k++)
1537 {
1538 double px = mx + (k ? h : -h) * (-uy);
1539 double py = my + (k ? h : -h) * ux;
1540 if (arc_contains_angle(e1, atan2(py - e1->cy, px - e1->cx)) &&
1541 arc_contains_angle(e2, atan2(py - e2->cy, px - e2->cx)))
1542 {
1543 ix[n] = px; iy[n] = py; n++;
1544 }
1545 /* A tangency has a single candidate point */
1546 if (h < MEOS_GEOM_TOLERANCE)
1547 break;
1548 }
1549 return n;
1550}

1636static inline bool
1637point_on_segment_within(double px, double py, double x1, double y1, double x2,
1638 double y2, double tol)
1639{
1640 /* Fast bounding-box rejection, which is where all but a few calls end */
1641 if ((px < Min(x1, x2) - tol) || (px > Max(x1, x2) + tol) ||
1642 (py < Min(y1, y2) - tol) || (py > Max(y1, y2) + tol))
1643 return false;
1644
1645 /* Vectors AP and AB */
1646 double apx = px - x1;
1647 double apy = py - y1;
1648 double abx = x2 - x1;
1649 double aby = y2 - y1;
1650
1651 /* The tolerance as the area it bounds over a segment of this size */
1652 double areatol = tol * (fabs(abx) + fabs(aby));
1653
1654 /* Collinearity check via cross product */
1655 double cross = apx * aby - apy * abx;
1656 if (fabs(cross) > areatol)
1657 return false;
1658
1659 /* Projection check via dot product */
1660 double dot = apx * abx + apy * aby;
1661 if (dot < -areatol)
1662 return false;
1663
1664 /* Check if P lies between A and B */
1665 double ab2 = abx * abx + aby * aby;
1666 if (dot > ab2 + areatol)
1667 return false;
1668 return true;
1669}
