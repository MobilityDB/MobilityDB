186static inline bool
187geom_ring_is_convex(const POINTARRAY *pa)
188{
189 if (! pa || pa->npoints < 4)
190 return false;
191 uint32_t n = pa->npoints - 1;
192 int sign = 0;
193 for (uint32_t i = 0; i < n; i++)
194 {
195 POINT4D a, b, c;
196 getPoint4d_p(pa, i, &a);
197 getPoint4d_p(pa, (i + 1) % n, &b);
198 getPoint4d_p(pa, (i + 2) % n, &c);
199 double turn = (b.x - a.x) * (c.y - b.y) - (b.y - a.y) * (c.x - b.x);
200 /* The tolerance liblwgeom uses for a coordinate comparison */
201 if (fabs(turn) <= MEOS_GEOM_TOLERANCE)
202 continue;
203 int current = turn > 0.0 ? 1 : -1;
204 if (sign == 0)
205 sign = current;
206 else if (sign != current)
207 return false;
208 }
209 return sign != 0;
210}

329static inline IntersectResult
330linesegm_intersect(double ax, double ay, double rx, double ry,
331 double cx, double cy, double dx, double dy)
332{
333 IntersectResult res = {INTERSECT_NONE, 0, 0};
334 double sx = dx - cx, sy = dy - cy; /* vector CD */
335 /* Where is the start of the second segment relative to the first? */
336 double qpx = cx - ax, qpy = cy - ay;
337
338 /* Are the two segments parallel? */
339 double rxs = rx * sy - ry * sx;
340
341 /* Collinear / parallel */
342 if (fabs(rxs) < MEOS_GEOM_TOLERANCE)
343 {
344 /* Is point C aligned with segment AB? */
345 double qpxr = qpx * ry - qpy * rx;
346 /* If qpxr != 0: parallel, if qpxr == 0: collinear */
347 if (fabs(qpxr) > MEOS_GEOM_TOLERANCE)
348 return res;
349
350 /* Collinear case */
351 double r2 = rx * rx + ry * ry;
352 if (r2 < MEOS_GEOM_TOLERANCE)
353 return res;
354
355 double t0 = (qpx * rx + qpy * ry) / r2;
356 double t1 = t0 + (sx * rx + sy * ry) / r2;
357
358 /* Order t0 < t1 */
359 if (t0 > t1) { double tmp = t0; t0 = t1; t1 = tmp; }
360 /* No intersection */
361 if (t1 < 0 || t0 > 1)
362 return res;
363
364 /* Clamp values */
365 if (t0 < 0) t0 = 0;
366 if (t1 > 1) t1 = 1;
367
368 if (fabs(t1 - t0) < MEOS_GEOM_TOLERANCE)
369 {
370 res.type = INTERSECT_POINT;
371 res.t0 = t0;
372 return res;
373 }
374
375 res.type = INTERSECT_OVERLAP;
376 res.t0 = t0;
377 res.t1 = t1;
378 return res;
379 }
380
381 /* Proper intersection */
382 double t = (qpx * sy - qpy * sx) / rxs;
383 double u = (qpx * ry - qpy * rx) / rxs;
384
385 if (t < -MEOS_GEOM_TOLERANCE || t > 1 + MEOS_GEOM_TOLERANCE ||
386 u < -MEOS_GEOM_TOLERANCE || u > 1 + MEOS_GEOM_TOLERANCE)
387 return res;
388
389 /* Clamp values */
390 if (fabs(t) < MEOS_GEOM_TOLERANCE) t = 0;
391 if (fabs(t - 1) < MEOS_GEOM_TOLERANCE) t = 1;
392
393 res.type = INTERSECT_POINT;
394 res.t0 = t;
395 return res;
396}

407static inline double
408arc_span_sweep(double theta0, double theta1, bool ccw)
409{
410 return ccw ?
411 angle_normalize(theta1 - theta0) :
412 angle_normalize(theta0 - theta1);
413}
414
419static inline double
420arc_span_offset(double theta0, bool ccw, double phi)
421{
422 return ccw ?
423 angle_normalize(phi - theta0) :
424 angle_normalize(theta0 - phi);
425}
426
434static inline bool
435arc_span_contains(double theta0, double theta1, bool ccw, double phi)
436{
437 double sweep = arc_span_sweep(theta0, theta1, ccw);
438 double off = arc_span_offset(theta0, ccw, phi);
439 return off <= sweep + MEOS_GEOM_TOLERANCE;
440}

445static inline bool
446arc_contains_angle(const Edge *e, double phi)
447{
448 return arc_span_contains(e->theta0, e->theta1, e->ccw, phi);
449}

456static inline void
457arc_set_bbox(Edge *e)
458{
459 double xmin = Min(e->x1, e->x2), xmax = Max(e->x1, e->x2);
460 double ymin = Min(e->y1, e->y2), ymax = Max(e->y1, e->y2);
461 const double ang[4] = {0.0, M_PI_2, M_PI, -M_PI_2};
462 const double ex[4] = {e->cx + e->radius, e->cx, e->cx - e->radius, e->cx};
463 const double ey[4] = {e->cy, e->cy + e->radius, e->cy, e->cy - e->radius};
464 for (int k = 0; k < 4; k++)
465 if (arc_contains_angle(e, ang[k]))
466 {
467 if (ex[k] < xmin) xmin = ex[k];
468 if (ex[k] > xmax) xmax = ex[k];
469 if (ey[k] < ymin) ymin = ey[k];
470 if (ey[k] > ymax) ymax = ey[k];
471 }
472 e->xmin = xmin; e->xmax = xmax; e->ymin = ymin; e->ymax = ymax;
473 return;
474}

479static inline bool
480point_on_arc(double px, double py, const Edge *e)
481{
482 /* The radial distance is read from the coordinates of the point and of the
483 * centre, so what it misses by is a property of their arithmetic and not of
484 * the arc: a few units in the last place of the largest of them. An absolute
485 * band asks a point at a projected 6.4e6 to sit on its own circle to a
486 * thousandth of what that coordinate can express, and the point the engine
487 * itself places on the arc then reads as lying off it. The edge carries the
488 * tolerance its own coordinates call for -- the same quantity
489 * #point_on_segment reads for a straight edge -- and the floor leaves the
490 * band at MEOS_GEOM_TOLERANCE where an edge carries none */
491 double d = hypot(px - e->cx, py - e->cy);
492 if (fabs(d - e->radius) > fmax(e->tol, MEOS_GEOM_TOLERANCE))
493 return false;
494 return arc_contains_angle(e, atan2(py - e->cy, px - e->cx));
495}

510static inline int
511arcsegm_intersect(double ax, double ay, double rx, double ry, const Edge *e,
512 double out[2])
513{
514 double aa = rx * rx + ry * ry;
515 /* Degenerate (zero-length) trajectory segment */
516 if (aa < MEOS_GEOM_TOLERANCE)
517 return 0;
518
519 double wx = ax - e->cx, wy = ay - e->cy;
520 double bb = 2 * (wx * rx + wy * ry);
521 double cc = wx * wx + wy * wy - e->radius * e->radius;
522 double disc = bb * bb - 4 * aa * cc;
523 /* The discriminant is the difference of two quantities of like size, so a
524 * segment touching the circle reaches zero only to their rounding. Reading
525 * it against that rounding rather than against zero keeps a touch one root:
526 * against zero it becomes two roots a square root of the rounding apart,
527 * which is a node pair far enough apart to be taken for two, and the sliver
528 * between them is not a piece of any boundary.
529 * Both terms VANISH where the segment starts on the circle and runs
530 * tangent to it there, which is what the offset of an edge leaving a vertex
531 * does against the round cap about that vertex, so a rounding read from them
532 * alone degenerates to zero and protects nothing. The quantities they are
533 * differences OF do not vanish: cc subtracts two of size radius squared, and
534 * bb is twice a product of size w by the segment direction */
535 double eps = 1e-14 * (fabs(bb * bb) + fabs(4 * aa * cc) +
536 aa * (wx * wx + wy * wy + e->radius * e->radius));
537 if (disc < -eps)
538 return 0;
539 if (disc <= eps)
540 disc = 0;
541
542 double sq = sqrt(disc);
543 double roots[2];
544 int nroots = 0;
545 roots[nroots++] = (-bb - sq) / (2 * aa);
546 /* Distinct second root only when the line is not tangent */
547 if (sq > MEOS_GEOM_TOLERANCE)
548 roots[nroots++] = (-bb + sq) / (2 * aa);
549
550 int n = 0;
551 for (int k = 0; k < nroots; k++)
552 {
553 double t = roots[k];
554 if (t < -MEOS_GEOM_TOLERANCE || t > 1 + MEOS_GEOM_TOLERANCE)
555 continue;
556 if (t < 0) t = 0;
557 if (t > 1) t = 1;
558 double px = ax + t * rx, py = ay + t * ry;
559 if (arc_contains_angle(e, atan2(py - e->cy, px - e->cx)))
560 out[n++] = t;
561 }
562 return n;
563}

573static inline double
574arc_point_parameter(const Edge *e, double x, double y)
575{
576 double phi = atan2(y - e->cy, x - e->cx);
577 double sweep = arc_span_sweep(e->theta0, e->theta1, e->ccw);
578 if (sweep < MEOS_GEOM_TOLERANCE)
579 return 0.0;
580 double off = arc_span_offset(e->theta0, e->ccw, phi);
581 double t = off / sweep;
582 /* A point the arithmetic puts just OUTSIDE the span belongs to whichever end
583 * it stands nearer, and one a rounding step BEFORE the start has turned
584 * nearly the whole circle rather than none of it */
585 if (t > 1.0 && (2 * M_PI - off) < (off - sweep))
586 return 0.0;
587 return (t < 0.0) ? 0.0 : ((t > 1.0) ? 1.0 : t);
588}

593static inline void
594arc_parameter_point(const Edge *e, double t, double *x, double *y)
595{
596 double sweep = arc_span_sweep(e->theta0, e->theta1, e->ccw);
597 double phi = e->ccw ? e->theta0 + t * sweep : e->theta0 - t * sweep;
598 *x = e->cx + e->radius * cos(phi);
599 *y = e->cy + e->radius * sin(phi);
600 return;
601}

614static inline int
615arcarc_intersect_points(const Edge *e1, const Edge *e2, double ix[2],
616 double iy[2], bool *same_circle)
617{
618 double dx = e2->cx - e1->cx, dy = e2->cy - e1->cy;
619 double d = hypot(dx, dy);
620 double r1 = e1->radius, r2 = e2->radius;
621 if (same_circle)
622 *same_circle = false;
623
624 /* Concentric supporting circles */
625 if (d < MEOS_GEOM_TOLERANCE)
626 {
627 if (fabs(r1 - r2) > MEOS_GEOM_TOLERANCE)
628 return 0;
629 /* Same circle: the arcs run along one another wherever their spans do,
630 * which is a shared stretch rather than a pair of points */
631 if (same_circle)
632 *same_circle = arc_contains_angle(e2, e1->theta0) ||
633 arc_contains_angle(e2, e1->theta1) ||
634 arc_contains_angle(e1, e2->theta0) ||
635 arc_contains_angle(e1, e2->theta1);
636 return 0;
637 }
638 /* Circles too far apart or one strictly inside the other */
639 if (d > r1 + r2 + MEOS_GEOM_TOLERANCE || d < fabs(r1 - r2) - MEOS_GEOM_TOLERANCE)
640 return 0;
641
642 double a = (d * d + r1 * r1 - r2 * r2) / (2 * d);
643 double h2 = r1 * r1 - a * a;
644 if (h2 < 0)
645 h2 = 0;
646 double h = sqrt(h2);
647 double ux = dx / d, uy = dy / d; /* Unit vector from c1 to c2 */
648 double mx = e1->cx + a * ux, my = e1->cy + a * uy; /* Foot on the centre line */
649
650 /* Candidate points m +/- h * perp(u), kept when both spans hold them */
651 int n = 0;
652 for (int k = 0; k < 2; k++)
653 {
654 double px = mx + (k ? h : -h) * (-uy);
655 double py = my + (k ? h : -h) * ux;
656 if (arc_contains_angle(e1, atan2(py - e1->cy, px - e1->cx)) &&
657 arc_contains_angle(e2, atan2(py - e2->cy, px - e2->cx)))
658 {
659 ix[n] = px; iy[n] = py; n++;
660 }
661 /* A tangency has a single candidate point */
662 if (h < MEOS_GEOM_TOLERANCE)
663 break;
664 }
665 return n;
666}

673static inline bool
674arcarc_intersect(const Edge *e1, const Edge *e2)

735static inline bool
736point_on_segment_within(double px, double py, double x1, double y1, double x2,
737 double y2, double tol)
738{
739 /* Fast bounding-box rejection, which is where all but a few calls end */
740 if ((px < fmin(x1, x2) - tol) || (px > fmax(x1, x2) + tol) ||
741 (py < fmin(y1, y2) - tol) || (py > fmax(y1, y2) + tol))
742 return false;
743
744 /* Vectors AP and AB */
745 double apx = px - x1;
746 double apy = py - y1;
747 double abx = x2 - x1;
748 double aby = y2 - y1;
749
750 /* The tolerance as the area it bounds over a segment of this size */
751 double areatol = tol * (fabs(abx) + fabs(aby));
752
753 /* Collinearity check via cross product */
754 double cross = apx * aby - apy * abx;
755 if (fabs(cross) > areatol)
756 return false;
757
758 /* Projection check via dot product */
759 double dot = apx * abx + apy * aby;
760 if (dot < -areatol)
761 return false;
762
763 /* Check if P lies between A and B */
764 double ab2 = abx * abx + aby * aby;
765 if (dot > ab2 + areatol)
766 return false;
767 return true;
768}

775static inline bool
776point_on_segment(double px, double py, double x1, double y1, double x2,
777 double y2)
